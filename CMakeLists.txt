cmake_minimum_required(VERSION 3.16)
project(atl-shelf VERSION 0.4.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)

set(CMAKE_AUTORCC ON)
add_executable(atl-shelf src/main.cpp)
target_sources(atl-shelf PRIVATE data/fdroid-signing-key.qrc)
target_link_libraries(atl-shelf PRIVATE Qt6::Widgets Qt6::Network)
find_package(KF6WindowSystem QUIET)
if(KF6WindowSystem_FOUND)
    target_link_libraries(atl-shelf PRIVATE KF6::WindowSystem)
    target_compile_definitions(atl-shelf PRIVATE HAVE_KWINDOWSYSTEM)
endif()
install(TARGETS atl-shelf RUNTIME DESTINATION bin)
install(FILES data/org.atl-shelf.desktop DESTINATION share/applications)
install(FILES data/org.atl-shelf.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES data/org.atl-shelf.png DESTINATION share/icons/hicolor/256x256/apps)

option(ATL_SHELF_TESTS "Build interaction tests" OFF)
if(ATL_SHELF_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(shelf-ux-tests tests/ux_test.cpp src/main.cpp data/fdroid-signing-key.qrc)
    target_compile_definitions(shelf-ux-tests PRIVATE ATL_SHELF_TESTING)
    target_link_libraries(shelf-ux-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
    add_test(NAME shelf-ux COMMAND shelf-ux-tests)
    set_tests_properties(shelf-ux PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()
